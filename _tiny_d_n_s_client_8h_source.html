190 bool resolve(String &io) {
191 size_t hostStart = 0;
192 int schemeEnd = io.indexOf("://");
193 if (schemeEnd >= 0) hostStart = (size_t)schemeEnd + 3;
194
195 size_t hostEnd = io.length();
196 int slashIdx = io.indexOf('/', hostStart);
197 int colonIdx = io.indexOf(':', hostStart);
198 if (slashIdx >= 0 && (size_t)slashIdx < hostEnd) hostEnd = (size_t)slashIdx;
199 if (colonIdx >= 0 && (size_t)colonIdx < hostEnd) hostEnd = (size_t)colonIdx;
200
201 if (hostEnd <= hostStart) return fail(DNSResult::NotConfigured);
202
203 String hostname = io.substring(hostStart, hostEnd);
204 IPAddress ip;
205 if (!resolve(hostname.c_str(), ip)) return false;
206
207 io = io.substring(0, hostStart) + ip.toString() + io.substring(hostEnd);
208 return true;
209 }

258 bool resolveType(const char *hostname, uint16_t qtype, uint8_t *rdataOut,
259 size_t rdataLen) {
260 if (!client_ || !hostname) return fail(DNSResult::NotConfigured);
261
262 if (!client_->connect(server_, port_)) return fail(DNSResult::ConnectFailed);
263
264 uint16_t id = nextId_++;
265 size_t len = buildQuery(buffer_, sizeof(buffer_), id, hostname, qtype);
266 if (len == 0) {
267 client_->stop();
268 return fail(DNSResult::InvalidResponse);
269 }
270
271 uint8_t lenPrefix[2] = {(uint8_t)(len >> 8), (uint8_t)(len & 0xFF)};
272 size_t written = client_->write(lenPrefix, 2);
273 written += client_->write(buffer_, len);
274 if (written != len + 2) {
275 client_->stop();
276 return fail(DNSResult::SendFailed);
277 }
278
279 bool ok = readAndParse(id, qtype, rdataOut, rdataLen);
280 client_->stop();
281 return ok;
282 }

295 size_t buildQuery(uint8_t *buf, size_t bufSize, uint16_t id,
296 const char *hostname, uint16_t qtype) {
297 size_t pos = 0;
298 if (bufSize < 12) return 0;
299
300 buf[pos++] = id >> 8;
301 buf[pos++] = id & 0xFF;
302 buf[pos++] = 0x01; // flags hi: recursion desired
303 buf[pos++] = 0x00; // flags lo
304 buf[pos++] = 0x00;
305 buf[pos++] = 0x01; // QDCOUNT = 1
306 buf[pos++] = 0x00;
307 buf[pos++] = 0x00; // ANCOUNT
308 buf[pos++] = 0x00;
309 buf[pos++] = 0x00; // NSCOUNT
310 buf[pos++] = 0x00;
311 buf[pos++] = 0x00; // ARCOUNT
312
313 const char *label = hostname;
314 while (*label) {
315 const char *dot = strchr(label, '.');
316 size_t labelLen = dot ? (size_t)(dot - label) : strlen(label);
317 if (labelLen == 0 || labelLen > 63) return 0;
318 if (pos + labelLen + 1 >= bufSize) return 0;
319 buf[pos++] = (uint8_t)labelLen;
320 memcpy(buf + pos, label, labelLen);
321 pos += labelLen;
322 label += labelLen;
323 if (*label == '.') label++;
324 }
325 if (pos + 1 >= bufSize) return 0;
326 buf[pos++] = 0x00; // root label
327
328 if (pos + 4 > bufSize) return 0;
329 buf[pos++] = qtype >> 8;
330 buf[pos++] = qtype & 0xFF;
331 buf[pos++] = 0x00;
332 buf[pos++] = 0x01; // QCLASS = IN
333
334 return pos;
335 }

346 bool readExact(uint8_t *dst, size_t n, uint32_t deadline) {
347 size_t got = 0;
348 while (got < n) {
349 int avail = client_->available();
350 if (avail > 0) {
351 int r = client_->read(dst + got, n - got);
352 if (r > 0) got += r;
353 } else if (!client_->connected() && client_->available() == 0) {
354 return false;
355 }
356 if (got < n && (int32_t)(millis() - deadline) >= 0) return false;
357 }
358 return true;
359 }

369 bool readAndParse(uint16_t expectedId, uint16_t qtype, uint8_t *rdataOut,
370 size_t rdataLen) {
371 uint32_t deadline = millis() + timeoutMs_;
372
373 uint8_t lenPrefix[2];
374 if (!readExact(lenPrefix, 2, deadline)) return fail(DNSResult::Timeout);
375 uint16_t msgLen = ((uint16_t)lenPrefix[0] << 8) | lenPrefix[1];
376 if (msgLen == 0 || msgLen > sizeof(buffer_)) {
377 return fail(DNSResult::InvalidResponse);
378 }
379 if (!readExact(buffer_, msgLen, deadline)) return fail(DNSResult::Timeout);
380
381 return parseResponse(buffer_, msgLen, expectedId, qtype, rdataOut, rdataLen);
382 }

392 static size_t skipName(const uint8_t *buf, size_t len, size_t pos) {
393 while (pos < len) {
394 uint8_t l = buf[pos];
395 if (l == 0) return pos + 1;
396 if ((l & 0xC0) == 0xC0) return pos + 2; // compressed pointer
397 pos += 1 + l;
398 }
399 return pos;
400 }

414 bool parseResponse(const uint8_t *buf, size_t len, uint16_t expectedId,
415 uint16_t qtype, uint8_t *rdataOut, size_t rdataLen) {
416 if (len < 12) return fail(DNSResult::InvalidResponse);
417
418 uint16_t id = ((uint16_t)buf[0] << 8) | buf[1];
419 if (id != expectedId) return fail(DNSResult::InvalidResponse);
420 if ((buf[2] & 0x80) == 0) return fail(DNSResult::InvalidResponse); // not a reply
421 if ((buf[3] & 0x0F) != 0) return fail(DNSResult::ServerError); // RCODE != 0
422
423 uint16_t qdcount = ((uint16_t)buf[4] << 8) | buf[5];
424 uint16_t ancount = ((uint16_t)buf[6] << 8) | buf[7];
425
426 size_t pos = 12;
427 for (uint16_t i = 0; i < qdcount; i++) {
428 pos = skipName(buf, len, pos);
429 pos += 4; // QTYPE + QCLASS
430 if (pos > len) return fail(DNSResult::InvalidResponse);
431 }
432
433 for (uint16_t i = 0; i < ancount; i++) {
434 pos = skipName(buf, len, pos);
435 if (pos + 10 > len) return fail(DNSResult::InvalidResponse);
436
437 uint16_t rtype = ((uint16_t)buf[pos] << 8) | buf[pos + 1];
438 uint32_t ttl = ((uint32_t)buf[pos + 4] << 24) | ((uint32_t)buf[pos + 5] << 16) |
439 ((uint32_t)buf[pos + 6] << 8) | buf[pos + 7];
440 pos += 8; // TYPE(2) + CLASS(2) + TTL(4)
441 uint16_t rdlen = ((uint16_t)buf[pos] << 8) | buf[pos + 1];
442 pos += 2;
443 if (pos + rdlen > len) return fail(DNSResult::InvalidResponse);
444
445 if (rtype == qtype && rdlen == rdataLen) {
446 memcpy(rdataOut, buf + pos, rdataLen);
447 lastResult_ = DNSResult::Ok;
448 lastTtl_ = ttl;
449 return true;
450 }
451 pos += rdlen;
452 }
453
454 return fail(DNSResult::NotFound);
455 }
