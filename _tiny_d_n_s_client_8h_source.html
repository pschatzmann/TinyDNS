161 bool resolve(String &io) {
162 size_t hostStart = 0;
163 int schemeEnd = io.indexOf("://");
164 if (schemeEnd >= 0) hostStart = (size_t)schemeEnd + 3;
165
166 size_t hostEnd = io.length();
167 int slashIdx = io.indexOf('/', hostStart);
168 int colonIdx = io.indexOf(':', hostStart);
169 if (slashIdx >= 0 && (size_t)slashIdx < hostEnd) hostEnd = (size_t)slashIdx;
170 if (colonIdx >= 0 && (size_t)colonIdx < hostEnd) hostEnd = (size_t)colonIdx;
171
172 if (hostEnd <= hostStart) return fail(DNSResult::NotConfigured);
173
174 String hostname = io.substring(hostStart, hostEnd);
175 IPAddress ip;
176 if (!resolve(hostname.c_str(), ip)) return false;
177
178 io = io.substring(0, hostStart) + ip.toString() + io.substring(hostEnd);
179 return true;
180 }

227 bool resolveType(const char *hostname, uint16_t qtype, uint8_t *rdataOut,
228 size_t rdataLen) {
229 if (!client_ || !hostname) return fail(DNSResult::NotConfigured);
230
231 if (!client_->connect(server_, port_)) return fail(DNSResult::ConnectFailed);
232
233 uint16_t id = nextId_++;
234 size_t len = buildQuery(buffer_, sizeof(buffer_), id, hostname, qtype);
235 if (len == 0) {
236 client_->stop();
237 return fail(DNSResult::InvalidResponse);
238 }
239
240 uint8_t lenPrefix[2] = {(uint8_t)(len >> 8), (uint8_t)(len & 0xFF)};
241 size_t written = client_->write(lenPrefix, 2);
242 written += client_->write(buffer_, len);
243 if (written != len + 2) {
244 client_->stop();
245 return fail(DNSResult::SendFailed);
246 }
247
248 bool ok = readAndParse(id, qtype, rdataOut, rdataLen);
249 client_->stop();
250 return ok;
251 }

264 size_t buildQuery(uint8_t *buf, size_t bufSize, uint16_t id,
265 const char *hostname, uint16_t qtype) {
266 size_t pos = 0;
267 if (bufSize < 12) return 0;
268
269 buf[pos++] = id >> 8;
270 buf[pos++] = id & 0xFF;
271 buf[pos++] = 0x01; // flags hi: recursion desired
272 buf[pos++] = 0x00; // flags lo
273 buf[pos++] = 0x00;
274 buf[pos++] = 0x01; // QDCOUNT = 1
275 buf[pos++] = 0x00;
276 buf[pos++] = 0x00; // ANCOUNT
277 buf[pos++] = 0x00;
278 buf[pos++] = 0x00; // NSCOUNT
279 buf[pos++] = 0x00;
280 buf[pos++] = 0x00; // ARCOUNT
281
282 const char *label = hostname;
283 while (*label) {
284 const char *dot = strchr(label, '.');
285 size_t labelLen = dot ? (size_t)(dot - label) : strlen(label);
286 if (labelLen == 0 || labelLen > 63) return 0;
287 if (pos + labelLen + 1 >= bufSize) return 0;
288 buf[pos++] = (uint8_t)labelLen;
289 memcpy(buf + pos, label, labelLen);
290 pos += labelLen;
291 label += labelLen;
292 if (*label == '.') label++;
293 }
294 if (pos + 1 >= bufSize) return 0;
295 buf[pos++] = 0x00; // root label
296
297 if (pos + 4 > bufSize) return 0;
298 buf[pos++] = qtype >> 8;
299 buf[pos++] = qtype & 0xFF;
300 buf[pos++] = 0x00;
301 buf[pos++] = 0x01; // QCLASS = IN
302
303 return pos;
304 }

315 bool readExact(uint8_t *dst, size_t n, uint32_t deadline) {
316 size_t got = 0;
317 while (got < n) {
318 int avail = client_->available();
319 if (avail > 0) {
320 int r = client_->read(dst + got, n - got);
321 if (r > 0) got += r;
322 } else if (!client_->connected() && client_->available() == 0) {
323 return false;
324 }
325 if (got < n && (int32_t)(millis() - deadline) >= 0) return false;
326 }
327 return true;
328 }

338 bool readAndParse(uint16_t expectedId, uint16_t qtype, uint8_t *rdataOut,
339 size_t rdataLen) {
340 uint32_t deadline = millis() + timeoutMs_;
341
342 uint8_t lenPrefix[2];
343 if (!readExact(lenPrefix, 2, deadline)) return fail(DNSResult::Timeout);
344 uint16_t msgLen = ((uint16_t)lenPrefix[0] << 8) | lenPrefix[1];
345 if (msgLen == 0 || msgLen > sizeof(buffer_)) {
346 return fail(DNSResult::InvalidResponse);
347 }
348 if (!readExact(buffer_, msgLen, deadline)) return fail(DNSResult::Timeout);
349
350 return parseResponse(buffer_, msgLen, expectedId, qtype, rdataOut, rdataLen);
351 }

361 static size_t skipName(const uint8_t *buf, size_t len, size_t pos) {
362 while (pos < len) {
363 uint8_t l = buf[pos];
364 if (l == 0) return pos + 1;
365 if ((l & 0xC0) == 0xC0) return pos + 2; // compressed pointer
366 pos += 1 + l;
367 }
368 return pos;
369 }

383 bool parseResponse(const uint8_t *buf, size_t len, uint16_t expectedId,
384 uint16_t qtype, uint8_t *rdataOut, size_t rdataLen) {
385 if (len < 12) return fail(DNSResult::InvalidResponse);
386
387 uint16_t id = ((uint16_t)buf[0] << 8) | buf[1];
388 if (id != expectedId) return fail(DNSResult::InvalidResponse);
389 if ((buf[2] & 0x80) == 0) return fail(DNSResult::InvalidResponse); // not a reply
390 if ((buf[3] & 0x0F) != 0) return fail(DNSResult::ServerError); // RCODE != 0
391
392 uint16_t qdcount = ((uint16_t)buf[4] << 8) | buf[5];
393 uint16_t ancount = ((uint16_t)buf[6] << 8) | buf[7];
394
395 size_t pos = 12;
396 for (uint16_t i = 0; i < qdcount; i++) {
397 pos = skipName(buf, len, pos);
398 pos += 4; // QTYPE + QCLASS
399 if (pos > len) return fail(DNSResult::InvalidResponse);
400 }
401
402 for (uint16_t i = 0; i < ancount; i++) {
403 pos = skipName(buf, len, pos);
404 if (pos + 10 > len) return fail(DNSResult::InvalidResponse);
405
406 uint16_t rtype = ((uint16_t)buf[pos] << 8) | buf[pos + 1];
407 uint32_t ttl = ((uint32_t)buf[pos + 4] << 24) | ((uint32_t)buf[pos + 5] << 16) |
408 ((uint32_t)buf[pos + 6] << 8) | buf[pos + 7];
409 pos += 8; // TYPE(2) + CLASS(2) + TTL(4)
410 uint16_t rdlen = ((uint16_t)buf[pos] << 8) | buf[pos + 1];
411 pos += 2;
412 if (pos + rdlen > len) return fail(DNSResult::InvalidResponse);
413
414 if (rtype == qtype && rdlen == rdataLen) {
415 memcpy(rdataOut, buf + pos, rdataLen);
416 lastResult_ = DNSResult::Ok;
417 lastTtl_ = ttl;
418 return true;
419 }
420 pos += rdlen;
421 }
422
423 return fail(DNSResult::NotFound);
424 }
