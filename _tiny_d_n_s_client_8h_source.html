152 bool resolve(String &io) {
153 size_t hostStart = 0;
154 int schemeEnd = io.indexOf("://");
155 if (schemeEnd >= 0) hostStart = (size_t)schemeEnd + 3;
156
157 size_t hostEnd = io.length();
158 int slashIdx = io.indexOf('/', hostStart);
159 int colonIdx = io.indexOf(':', hostStart);
160 if (slashIdx >= 0 && (size_t)slashIdx < hostEnd) hostEnd = (size_t)slashIdx;
161 if (colonIdx >= 0 && (size_t)colonIdx < hostEnd) hostEnd = (size_t)colonIdx;
162
163 if (hostEnd <= hostStart) return fail(DNSResult::NotConfigured);
164
165 String hostname = io.substring(hostStart, hostEnd);
166 IPAddress ip;
167 if (!resolve(hostname.c_str(), ip)) return false;
168
169 io = io.substring(0, hostStart) + ip.toString() + io.substring(hostEnd);
170 return true;
171 }

218 bool resolveType(const char *hostname, uint16_t qtype, uint8_t *rdataOut,
219 size_t rdataLen) {
220 if (!client_ || !hostname) return fail(DNSResult::NotConfigured);
221
222 if (!client_->connect(server_, port_)) return fail(DNSResult::ConnectFailed);
223
224 uint16_t id = nextId_++;
225 size_t len = buildQuery(buffer_, sizeof(buffer_), id, hostname, qtype);
226 if (len == 0) {
227 client_->stop();
228 return fail(DNSResult::InvalidResponse);
229 }
230
231 uint8_t lenPrefix[2] = {(uint8_t)(len >> 8), (uint8_t)(len & 0xFF)};
232 size_t written = client_->write(lenPrefix, 2);
233 written += client_->write(buffer_, len);
234 if (written != len + 2) {
235 client_->stop();
236 return fail(DNSResult::SendFailed);
237 }
238
239 bool ok = readAndParse(id, qtype, rdataOut, rdataLen);
240 client_->stop();
241 return ok;
242 }

255 size_t buildQuery(uint8_t *buf, size_t bufSize, uint16_t id,
256 const char *hostname, uint16_t qtype) {
257 size_t pos = 0;
258 if (bufSize < 12) return 0;
259
260 buf[pos++] = id >> 8;
261 buf[pos++] = id & 0xFF;
262 buf[pos++] = 0x01; // flags hi: recursion desired
263 buf[pos++] = 0x00; // flags lo
264 buf[pos++] = 0x00;
265 buf[pos++] = 0x01; // QDCOUNT = 1
266 buf[pos++] = 0x00;
267 buf[pos++] = 0x00; // ANCOUNT
268 buf[pos++] = 0x00;
269 buf[pos++] = 0x00; // NSCOUNT
270 buf[pos++] = 0x00;
271 buf[pos++] = 0x00; // ARCOUNT
272
273 const char *label = hostname;
274 while (*label) {
275 const char *dot = strchr(label, '.');
276 size_t labelLen = dot ? (size_t)(dot - label) : strlen(label);
277 if (labelLen == 0 || labelLen > 63) return 0;
278 if (pos + labelLen + 1 >= bufSize) return 0;
279 buf[pos++] = (uint8_t)labelLen;
280 memcpy(buf + pos, label, labelLen);
281 pos += labelLen;
282 label += labelLen;
283 if (*label == '.') label++;
284 }
285 if (pos + 1 >= bufSize) return 0;
286 buf[pos++] = 0x00; // root label
287
288 if (pos + 4 > bufSize) return 0;
289 buf[pos++] = qtype >> 8;
290 buf[pos++] = qtype & 0xFF;
291 buf[pos++] = 0x00;
292 buf[pos++] = 0x01; // QCLASS = IN
293
294 return pos;
295 }

306 bool readExact(uint8_t *dst, size_t n, uint32_t deadline) {
307 size_t got = 0;
308 while (got < n) {
309 int avail = client_->available();
310 if (avail > 0) {
311 int r = client_->read(dst + got, n - got);
312 if (r > 0) got += r;
313 } else if (!client_->connected() && client_->available() == 0) {
314 return false;
315 }
316 if (got < n && (int32_t)(millis() - deadline) >= 0) return false;
317 }
318 return true;
319 }

329 bool readAndParse(uint16_t expectedId, uint16_t qtype, uint8_t *rdataOut,
330 size_t rdataLen) {
331 uint32_t deadline = millis() + timeoutMs_;
332
333 uint8_t lenPrefix[2];
334 if (!readExact(lenPrefix, 2, deadline)) return fail(DNSResult::Timeout);
335 uint16_t msgLen = ((uint16_t)lenPrefix[0] << 8) | lenPrefix[1];
336 if (msgLen == 0 || msgLen > sizeof(buffer_)) {
337 return fail(DNSResult::InvalidResponse);
338 }
339 if (!readExact(buffer_, msgLen, deadline)) return fail(DNSResult::Timeout);
340
341 return parseResponse(buffer_, msgLen, expectedId, qtype, rdataOut, rdataLen);
342 }

352 static size_t skipName(const uint8_t *buf, size_t len, size_t pos) {
353 while (pos < len) {
354 uint8_t l = buf[pos];
355 if (l == 0) return pos + 1;
356 if ((l & 0xC0) == 0xC0) return pos + 2; // compressed pointer
357 pos += 1 + l;
358 }
359 return pos;
360 }

374 bool parseResponse(const uint8_t *buf, size_t len, uint16_t expectedId,
375 uint16_t qtype, uint8_t *rdataOut, size_t rdataLen) {
376 if (len < 12) return fail(DNSResult::InvalidResponse);
377
378 uint16_t id = ((uint16_t)buf[0] << 8) | buf[1];
379 if (id != expectedId) return fail(DNSResult::InvalidResponse);
380 if ((buf[2] & 0x80) == 0) return fail(DNSResult::InvalidResponse); // not a reply
381 if ((buf[3] & 0x0F) != 0) return fail(DNSResult::ServerError); // RCODE != 0
382
383 uint16_t qdcount = ((uint16_t)buf[4] << 8) | buf[5];
384 uint16_t ancount = ((uint16_t)buf[6] << 8) | buf[7];
385
386 size_t pos = 12;
387 for (uint16_t i = 0; i < qdcount; i++) {
388 pos = skipName(buf, len, pos);
389 pos += 4; // QTYPE + QCLASS
390 if (pos > len) return fail(DNSResult::InvalidResponse);
391 }
392
393 for (uint16_t i = 0; i < ancount; i++) {
394 pos = skipName(buf, len, pos);
395 if (pos + 10 > len) return fail(DNSResult::InvalidResponse);
396
397 uint16_t rtype = ((uint16_t)buf[pos] << 8) | buf[pos + 1];
398 uint32_t ttl = ((uint32_t)buf[pos + 4] << 24) | ((uint32_t)buf[pos + 5] << 16) |
399 ((uint32_t)buf[pos + 6] << 8) | buf[pos + 7];
400 pos += 8; // TYPE(2) + CLASS(2) + TTL(4)
401 uint16_t rdlen = ((uint16_t)buf[pos] << 8) | buf[pos + 1];
402 pos += 2;
403 if (pos + rdlen > len) return fail(DNSResult::InvalidResponse);
404
405 if (rtype == qtype && rdlen == rdataLen) {
406 memcpy(rdataOut, buf + pos, rdataLen);
407 lastResult_ = DNSResult::Ok;
408 lastTtl_ = ttl;
409 return true;
410 }
411 pos += rdlen;
412 }
413
414 return fail(DNSResult::NotFound);
415 }
