149 TinyDNSBlockAction check(const char *name) const {
150 if (!name || (active_ == 0 && !store_)) return TinyDNSBlockAction::None;
151 char buf[kMaxNameLen + 1];
152 size_t len = 0;
153 if (!normalize(name, buf, sizeof(buf), len)) return TinyDNSBlockAction::None;
154
155 size_t pos = 0;
156 while (true) {
157 const char *suffix = buf + pos;
158 const Slot *s = findSlot(suffix, hashString(suffix));
159 if (s && s->flags) {
160 return (s->flags & kFlagAllow) ? TinyDNSBlockAction::Allow
161 : TinyDNSBlockAction::Block;
162 }
163 if (store_ && store_->contains(suffix)) return TinyDNSBlockAction::Block;
164 const char *dot = strchr(suffix, '.');
165 if (!dot) break;
166 pos = (size_t)(dot - buf) + 1;
167 }
168 return TinyDNSBlockAction::None;
169 }

210 static uint32_t hashString(const char *s) {
211 uint32_t h = 2166136261UL; // FNV-1a
212 for (; *s; s++) {
213 h ^= (uint8_t)*s;
214 h *= 16777619UL;
215 }
216 return h;

227 static bool normalize(const char *in, char *out, size_t cap, size_t &outLen) {
228 size_t n = strlen(in);
229 if (n > 0 && in[n - 1] == '.') n--;
230 if (n == 0 || n >= cap) return false;
231 for (size_t i = 0; i < n; i++) {
232 char c = in[i];
233 if (c >= 'A' && c <= 'Z') c = c + 32;
234 if (c == '.') {
235 if (i == 0 || in[i - 1] == '.') return false;
236 } else if (!isNameChar(c)) {
237 return false;
238 }
239 out[i] = c;
240 }
241 out[n] = 0;
242 outLen = n;
243 return true;
244 }

246 Slot *findSlot(const char *name, uint32_t h) const {
247 if (!slots_) return nullptr;
248 size_t mask = cap_ - 1;
249 size_t i = h & mask;
250 while (slots_[i].off != kEmptyOff) {
251 Slot &s = slots_[i];
252 if (s.hash == h && strcmp(pool_ + s.off, name) == 0) return &s;
253 i = (i + 1) & mask;
254 }
255 return nullptr;
256 }

258 static void insertRaw(Slot *table, size_t cap, const Slot &s) {
259 size_t mask = cap - 1;
260 size_t i = s.hash & mask;
261 while (table[i].off != kEmptyOff) i = (i + 1) & mask;
262 table[i] = s;
263 }

266 bool reserveSlot() {
267 if (cap_ && (used_ + 1) * 4 <= cap_ * 3) return true;
268 size_t newCap = cap_ ? cap_ * 2 : 64;
269 Slot *table = (Slot *)tinyDnsAlloc(newCap * sizeof(Slot));
270 if (!table) return false;
271 for (size_t i = 0; i < newCap; i++) table[i].off = kEmptyOff;
272 for (size_t i = 0; i < cap_; i++) {
273 if (slots_[i].off != kEmptyOff) insertRaw(table, newCap, slots_[i]);
274 }
275 tinyDnsFree(slots_);
276 slots_ = table;
277 cap_ = newCap;
278 return true;
279 }

281 bool intern(const char *s, size_t len, uint32_t &offOut) {
282 size_t need = poolLen_ + len + 1;
283 if (need > poolCap_) {
284 // Grow by copying: PSRAM blocks cannot be resized with realloc().
285 size_t newCap = poolCap_ ? poolCap_ * 2 : 1024;
286 while (newCap < need) newCap *= 2;
287 char *p = (char *)tinyDnsAlloc(newCap);
288 if (!p) return false;
289 if (poolLen_) memcpy(p, pool_, poolLen_);
290 tinyDnsFree(pool_);
291 pool_ = p;
292 poolCap_ = newCap;
293 }
294 memcpy(pool_ + poolLen_, s, len);
295 pool_[poolLen_ + len] = 0;
296 offOut = (uint32_t)poolLen_;
297 poolLen_ = need;
298 return true;
299 }

301 bool addRule(const char *domain, uint8_t flag) {
302 char buf[kMaxNameLen + 1];
303 size_t len = 0;
304 if (!domain || !normalize(domain, buf, sizeof(buf), len)) return false;
305 uint32_t h = hashString(buf);
306 Slot *s = findSlot(buf, h);
307 if (s) {
308 if (s->flags == 0) active_++;
309 s->flags |= flag;
310 return true;
311 }
312 if (!reserveSlot()) return false;
313 uint32_t off = 0;
314 if (!intern(buf, len, off)) return false;
315 Slot ns = {h, off, flag};
316 insertRaw(slots_, cap_, ns);
317 used_++;
318 active_++;
319 return true;
320 }

341 for (; *a && *b; a++, b++) {

365 static char *nextToken(char **p) {
366 char *s = *p;
367 while (*s == ' ' || *s == '\t' || *s == '\r' || *s == '\n') s++;
368 if (*s == 0) {
369 *p = s;
370 return nullptr;
371 }
372 char *start = s;
373 while (*s && *s != ' ' && *s != '\t' && *s != '\r' && *s != '\n') s++;
374 if (*s) *s++ = 0;
375 *p = s;
376 return start;
377 }

385 size_t addAdblockRule(char *s, uint8_t flag) {
386 size_t n = strlen(s);

393 size_t processLine(char *line, uint8_t flag) {
394 char *p = line;
395 while (*p == ' ' || *p == '\t') p++;
396 size_t n = strlen(p);
397 while (n > 0 && (p[n - 1] == ' ' || p[n - 1] == '\t' || p[n - 1] == '\r')) {
398 p[--n] = 0;
399 }
400 if (*p == 0 || *p == '!' || *p == '[') return 0; // AdBlock comment/header
401
402 if (strncmp(p, "@@||", 4) == 0) return addAdblockRule(p + 4, kFlagAllow);
403 if (strncmp(p, "||", 2) == 0) return addAdblockRule(p + 2, flag);
404
405 // Cosmetic filters (example.com##.ad) are not domain rules; without this
406 // check the '#' stripping below would turn them into blocks.
407 if (strstr(p, "##") || strstr(p, "#@#") || strstr(p, "#?#")) return 0;
408
409 char *hash = strchr(p, '#');
410 if (hash) *hash = 0;
411
412 char *cursor = p;
413 char *tok = nextToken(&cursor);
414 if (!tok) return 0;
415
416 if (isIpLiteral(tok)) {
417 // Hosts format: only sinkhole addresses mark a line as a block entry
418 if (!isSinkholeAddress(tok)) return 0;
419 tok = nextToken(&cursor);
420 }
421
422 size_t added = 0;
423 for (; tok; tok = nextToken(&cursor)) {
424 if (tok[0] == '*' && tok[1] == '.') tok += 2; // "*.ads.com" -> "ads.com"
425 if (isReservedName(tok)) continue;
426 if (addRule(tok, flag)) added++;
427 }
428 return added;
429 }

431 template <typename StreamT>
432 size_t parseStream(StreamT &in, uint8_t flag) {
433 char line[kMaxLineLen];
434 size_t n = 0;
435 bool tooLong = false;
436 size_t added = 0;
437
438 while (in.available() > 0) {
439 int c = in.read();
440 if (c < 0) break;
441 if (c == '\n') {
442 if (!tooLong) {
443 line[n] = 0;
444 added += processLine(line, flag);
445 }
446 n = 0;
447 tooLong = false;
448 } else if (n < kMaxLineLen - 1) {
449 line[n++] = (char)c;
450 } else {
451 tooLong = true;
452 }
453 }
454 // Last line without a trailing newline
455 if (!tooLong && n > 0) {
456 line[n] = 0;
457 added += processLine(line, flag);
458 }
459 return added;
460 }
