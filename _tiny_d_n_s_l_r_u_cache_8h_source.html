146 bool put(const char *name, uint16_t qtype, const uint8_t *rdata,
147 uint8_t rdlen, uint32_t expiresMs) {
148 if (!entries_) return false;
149 size_t nlen = strlen(name);
150 if (nlen == 0 || nlen >= kMaxNameLen || rdlen == 0 || rdlen > 16) {
151 return false;
152 }
153
154 int i = find(name, qtype);
155 if (i >= 0) {
156 unlink(i);
157 } else if (count_ < capacity_) {
158 i = firstFree();
159 count_++;
160 } else {
161 i = tail_; // evict least recently used
162 unlink(i);
163 }
164
165 Entry &e = entries_[i];
166 for (size_t k = 0; k < nlen; k++) e.name[k] = lower(name[k]);
167 e.name[nlen] = 0;
168 e.qtype = qtype;
169 memcpy(e.rdata, rdata, rdlen);
170 e.rdlen = rdlen;
171 e.expiresMs = expiresMs;
172 e.used = true;
173 pushFront(i);
174 return true;
175 }
