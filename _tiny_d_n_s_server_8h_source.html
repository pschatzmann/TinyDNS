130 bool handle(Client &client) {
131 uint32_t deadline = millis() + timeoutMs_;
132
133 uint8_t lenPrefix[2];
134 if (!readExact(client, lenPrefix, 2, deadline)) return false;
135 size_t msgLen = ((size_t)lenPrefix[0] << 8) | lenPrefix[1];
136 if (msgLen < 12 || msgLen > sizeof(query_)) return false;
137 if (!readExact(client, query_, msgLen, deadline)) return false;
138
139 uint16_t id = ((uint16_t)query_[0] << 8) | query_[1];
140 if (query_[2] & 0x80) return false; // not a query, ignore
141 uint8_t opcode = (query_[2] >> 3) & 0x0F;
142 uint16_t qdcount = ((uint16_t)query_[4] << 8) | query_[5];
143
144 if (opcode != 0) return sendReply(client, id, RCODE_NOTIMP, 0, 0, nullptr, 0, 0);
145 if (qdcount != 1) return sendReply(client, id, RCODE_FORMERR, 0, 0, nullptr, 0, 0);
146
147 char name[Cache::kMaxNameLen];
148 uint16_t qtype = 0;
149 size_t qEnd = 0;
150 if (!parseQuestion(msgLen, name, sizeof(name), qtype, qEnd)) {
151 return sendReply(client, id, RCODE_FORMERR, 0, 0, nullptr, 0, 0);
152 }
153 size_t questionLen = qEnd - 12;
154
155 if (qtype != kTypeA && qtype != kTypeAAAA) {
156 return sendReply(client, id, RCODE_NOTIMP, questionLen, qtype, nullptr, 0, 0);
157 }
158
159 // Static host table: authoritative, bypasses cache and upstream
160 const TinyDNSHost *host = findHost(name);
161 if (host) {
162 if (qtype == kTypeA) {
163 uint8_t ip4[4] = {host->ip[0], host->ip[1], host->ip[2], host->ip[3]};
164 return sendReply(client, id, RCODE_OK, questionLen, qtype, ip4, 4, maxTtl_);
165 }
166 // Host exists but has no IPv6 address: NODATA (NOERROR, no answer)
167 return sendReply(client, id, RCODE_OK, questionLen, qtype, nullptr, 0, 0);
168 }
169
170 // Cache lookup
171 uint32_t now = millis();
172 const Cache::Entry *e = cache_.get(name, qtype, now);
173 if (e) {
174 hits_++;
175 uint32_t remaining = (uint32_t)(e->expiresMs - now) / 1000UL;
176 return sendReply(client, id, RCODE_OK, questionLen, qtype, e->rdata,
177 e->rdlen, remaining);
178 }
179
180 // Cache miss: ask upstream
181 misses_++;
182 uint8_t rdata[16];
183 uint8_t rdlen = 0;
184 bool ok = false;
185 if (qtype == kTypeA) {
186 IPAddress ip;
187 ok = upstream_.resolve(name, ip);
188 if (ok) {
189 for (int k = 0; k < 4; k++) rdata[k] = ip[k];
190 rdlen = 4;
191 }
192 } else {
193 ok = upstream_.resolveV6(name, rdata);
194 if (ok) rdlen = 16;
195 }
196
197 if (!ok) {
198 uint8_t rcode = upstream_.lastResult() == DNSResult::NotFound
199 ? RCODE_NXDOMAIN
200 : RCODE_SERVFAIL;
201 return sendReply(client, id, rcode, questionLen, qtype, nullptr, 0, 0);
202 }
203
204 uint32_t ttl = upstream_.lastTtl();
205 if (ttl > maxTtl_) ttl = maxTtl_;
206 if (ttl > 0) {
207 cache_.put(name, qtype, rdata, rdlen, millis() + ttl * 1000UL);
208 }
209 return sendReply(client, id, RCODE_OK, questionLen, qtype, rdata, rdlen, ttl);
210 }

237 const TinyDNSHost *findHost(const char *name) const {
238 if (!hosts_) return nullptr;
239 for (size_t i = 0; i < hostCount_; i++) {
240 const char *a = hosts_[i].name;
241 const char *b = name;
242 if (!a) continue;
243 while (*a && *b && lowerChar(*a) == lowerChar(*b)) {
244 a++;
245 b++;
246 }
247 if (*a == 0 && *b == 0) return &hosts_[i];
248 }
249 return nullptr;
250 }

257 static bool readExact(Client &client, uint8_t *dst, size_t n, uint32_t deadline) {
258 size_t got = 0;
259 while (got < n) {
260 int avail = client.available();
261 if (avail > 0) {
262 int r = client.read(dst + got, n - got);
263 if (r > 0) got += r;
264 } else if (!client.connected()) {
265 return false;
266 }
267 if (got < n && (int32_t)(millis() - deadline) >= 0) return false;
268 }
269 return true;
270 }

281 bool parseQuestion(size_t msgLen, char *name, size_t nameCap, uint16_t &qtype,
282 size_t &qEnd) const {
283 size_t pos = 12;
284 size_t nlen = 0;
285 while (true) {
286 if (pos >= msgLen) return false;
287 uint8_t l = query_[pos];
288 if (l == 0) {
289 pos++;
290 break;
291 }
292 if ((l & 0xC0) || l > 63) return false; // compression not expected here
293 if (pos + 1 + l > msgLen) return false;
294 if (nlen + l + 1 >= nameCap) return false;
295 if (nlen > 0) name[nlen++] = '.';
296 memcpy(name + nlen, query_ + pos + 1, l);
297 nlen += l;
298 pos += 1 + l;
299 }
300 if (nlen == 0) return false;
301 name[nlen] = 0;
302 if (pos + 4 > msgLen) return false;
303 qtype = ((uint16_t)query_[pos] << 8) | query_[pos + 1];
304 qEnd = pos + 4;
305 return true;
306 }

314 bool sendReply(Client &client, uint16_t id, uint8_t rcode, size_t questionLen,
315 uint16_t qtype, const uint8_t *rdata, uint8_t rdlen,
316 uint32_t ttl) {
317 bool hasAnswer = questionLen > 0 && rdata != nullptr && rdlen > 0;
318 size_t pos = 0;
319
320 reply_[pos++] = id >> 8;
321 reply_[pos++] = id & 0xFF;
322 reply_[pos++] = 0x81; // QR=1 (response), RD=1
323 reply_[pos++] = 0x80 | (rcode & 0x0F); // RA=1, RCODE
324 reply_[pos++] = 0x00;
325 reply_[pos++] = questionLen > 0 ? 1 : 0; // QDCOUNT
326 reply_[pos++] = 0x00;
327 reply_[pos++] = hasAnswer ? 1 : 0; // ANCOUNT
328 reply_[pos++] = 0x00;
329 reply_[pos++] = 0x00; // NSCOUNT
330 reply_[pos++] = 0x00;
331 reply_[pos++] = 0x00; // ARCOUNT
332
333 if (questionLen > 0) {
334 memcpy(reply_ + pos, query_ + 12, questionLen);
335 pos += questionLen;
336 }
337
338 if (hasAnswer) {
339 reply_[pos++] = 0xC0; // name: pointer to question at offset 12
340 reply_[pos++] = 0x0C;
341 reply_[pos++] = qtype >> 8;
342 reply_[pos++] = qtype & 0xFF;
343 reply_[pos++] = 0x00; // CLASS = IN
344 reply_[pos++] = 0x01;
345 reply_[pos++] = (uint8_t)(ttl >> 24);
346 reply_[pos++] = (uint8_t)(ttl >> 16);
347 reply_[pos++] = (uint8_t)(ttl >> 8);
348 reply_[pos++] = (uint8_t)(ttl & 0xFF);
349 reply_[pos++] = 0x00; // RDLENGTH
350 reply_[pos++] = rdlen;
351 memcpy(reply_ + pos, rdata, rdlen);
352 pos += rdlen;
353 }
354
355 uint8_t lenPrefix[2] = {(uint8_t)(pos >> 8), (uint8_t)(pos & 0xFF)};
356 size_t written = client.write(lenPrefix, 2);
357 written += client.write(reply_, pos);
358 client.flush();
359 return written == pos + 2;
360 }
