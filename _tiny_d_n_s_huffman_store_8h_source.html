105 bool contains(const char *domain) const {
106 if (!ready_ || !domain) return false;
107 char key[kMaxKeyLen];
108 size_t keyLen = 0;
109 if (!reverseKey(domain, key, sizeof(key), keyLen)) return false;
110
111 // Find the number of blocks whose first key is <= key.
112 uint32_t lo = 0, hi = d_.blockCount;
113 char first[kMaxKeyLen];
114 while (lo < hi) {
115 uint32_t mid = lo + (hi - lo) / 2;
116 size_t firstLen = 0;
117 if (!firstKey(mid, first, sizeof(first), firstLen)) return false;
118 if (strcmp(first, key) <= 0) {
119 lo = mid + 1;
120 } else {
121 hi = mid;
122 }
123 }
124 if (lo == 0) return false; // key sorts before the first domain
125
126 // Scan the one block that can contain key.
127 uint32_t block = lo - 1;
128 Cursor cur = cursorFor(block);
129 uint32_t remaining = recordsInBlock(block);
130 char candidate[kMaxKeyLen];
131 size_t candLen = 0;
132 for (uint32_t i = 0; i < remaining; i++) {
133 if (!nextKey(cur, candidate, sizeof(candidate), candLen)) return false;
134 int cmp = strcmp(candidate, key);
135 if (cmp == 0) return true;
136 if (cmp > 0) return false;
137 }
138 return false;
139 }

179 static uint32_t readBE32(const uint8_t *p) {
180 return ((uint32_t)TINY_DNS_PGM_READ8(p) << 24) |
181 ((uint32_t)TINY_DNS_PGM_READ8(p + 1) << 16) |
182 ((uint32_t)TINY_DNS_PGM_READ8(p + 2) << 8) |

193 bool buildTable(const uint8_t *lens, Table &t) const {
194 memset(t.count, 0, sizeof(t.count));
195 uint32_t used = 0;
196 for (int sym = 0; sym < 256; sym++) {
197 uint8_t len = TINY_DNS_PGM_READ8(lens + sym);
198 if (len > kMaxCodeLen) return false;
199 if (len) {
200 t.count[len]++;
201 used++;
202 }
203 }
204 if (used == 0) return false;
205
206 // Reject over-subscribed codes (Kraft sum above 1).
207 int left = 1;
208 for (int len = 1; len <= kMaxCodeLen; len++) {
209 left <<= 1;
210 left -= t.count[len];
211 if (left < 0) return false;
212 }
213
214 uint16_t offs[kMaxCodeLen + 2];
215 offs[1] = 0;
216 for (int len = 1; len < kMaxCodeLen; len++) offs[len + 1] = offs[len] + t.count[len];
217 for (int sym = 0; sym < 256; sym++) {
218 uint8_t len = TINY_DNS_PGM_READ8(lens + sym);
219 if (len) t.symbol[offs[len]++] = (uint8_t)sym;
220 }
221 return true;
222 }

225 int decodeSymbol(const Table &t, BitReader &r) const {
226 int code = 0, first = 0, index = 0;
227 for (int len = 1; len <= kMaxCodeLen; len++) {
228 code |= r.readBit();
229 int count = t.count[len];
230 if (code - first < count) {
231 if (r.error) return -1;
232 return t.symbol[index + (code - first)];
233 }
234 index += count;
235 first += count;
236 first <<= 1;
237 code <<= 1;
238 }
239 return -1;
240 }

242 Cursor cursorFor(uint32_t block) const {
243 const uint8_t *entry = d_.index + (size_t)block * 8;
244 Cursor c;
245 c.prefix = {d_.prefixBits, readBE32(entry), d_.prefixBitCount, false};
246 c.suffix = {d_.suffixBits, readBE32(entry + 4), d_.suffixBitCount, false};
247 return c;
248 }

254 bool nextKey(Cursor &c, char *key, size_t cap, size_t &len) const {
255 int shared = decodeSymbol(prefix_, c.prefix);
256 if (shared < 0 || (size_t)shared > len) return false;
257 len = (size_t)shared;
258 while (true) {
259 int b = decodeSymbol(suffix_, c.suffix);
260 if (b < 0) return false;
261 if (b == 0) break;
262 if (len + 1 >= cap) return false;
263 key[len++] = (char)b;
264 }
265 key[len] = 0;
266 return true;
267 }

283 static bool reverseKey(const char *domain, char *out, size_t cap, size_t &outLen) {
284 size_t n = strlen(domain);
285 if (n > 0 && domain[n - 1] == '.') n--;
286 if (n == 0 || n >= cap) return false;
287
288 size_t out_pos = 0;
289 size_t end = n;
290 while (true) {
291 size_t start = end;
292 while (start > 0 && domain[start - 1] != '.') start--;
293 if (start == end) return false; // empty label
294 if (out_pos > 0) out[out_pos++] = '.';
295 for (size_t i = start; i < end; i++) {
296 char c = domain[i];
297 if (c >= 'A' && c <= 'Z') c = c + 32;
298 if (!isNameChar(c)) return false;
299 out[out_pos++] = c;
300 }
301 if (start == 0) break;
302 end = start - 1; // skip the dot
303 }
304 out[out_pos] = 0;
305 outLen = out_pos;
306 return true;
307 }
