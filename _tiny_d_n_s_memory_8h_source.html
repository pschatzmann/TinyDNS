39inline void *tinyDnsAlloc(size_t bytes) {
40#if defined(TINY_DNS_USE_PSRAM)
41 void *p = heap_caps_calloc(1, bytes, MALLOC_CAP_SPIRAM);
42 if (p) return p; // falls through to internal RAM if PSRAM is unavailable
43#elif defined(TINY_DNS_USE_PSRAM_PICO)
44 // arduino-pico: pmalloc() allocates from PSRAM; release with free().
45 void *p = pmalloc(bytes);
46 if (p) {
47 memset(p, 0, bytes);
48 return p;
49 }
50#endif
51 return calloc(1, bytes);
52}
